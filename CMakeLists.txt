cmake_minimum_required(VERSION 3.16)
project(Dots_and_Boxes)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_INTERPROCEDURAL_OPTIMIZATION_RELEASE ON)

set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTOUIC ON)
set(CMAKE_AUTORCC ON)

find_package(Qt6 REQUIRED COMPONENTS Core Gui Widgets)
find_package(TBB REQUIRED)

if(NOT MaxBoardSize)
  set(MaxBoardSize 36)
endif()
message(STATUS "MaxBoardSize: ${MaxBoardSize}")

set(ALL_LIBRARIES)
foreach(BoardSize RANGE 1 ${MaxBoardSize})
  add_subdirectory(src "source/src/src${BoardSize}x${BoardSize}")
  add_subdirectory(frontend "source/frontend/frontend${BoardSize}x${BoardSize}")
  add_subdirectory(extern "source/extern/extern${BoardSize}x${BoardSize}")
  list(APPEND ALL_LIBRARIES ${ModelLibraryName} ${BoardLibraryName} ${RobotLibraryName} ${FrontendLibraryName} ${ExternLibraryName})
endforeach()

qt_add_library("${PROJECT_NAME}_static" STATIC main.cpp)
target_include_directories("${PROJECT_NAME}_static" PRIVATE "${CMAKE_SOURCE_DIR}/include")
target_compile_definitions("${PROJECT_NAME}_static" PRIVATE QT_NO_KEYWORDS __MaxBoardSize__=${MaxBoardSize})
target_link_libraries("${PROJECT_NAME}_static" PUBLIC Qt6::Core Qt6::Gui Qt6::Widgets ${ALL_LIBRARIES})

qt_add_executable("${PROJECT_NAME}")
target_include_directories("${PROJECT_NAME}" PRIVATE "${CMAKE_SOURCE_DIR}/include")
target_compile_definitions("${PROJECT_NAME}" PRIVATE QT_NO_KEYWORDS __MaxBoardSize__=${MaxBoardSize})
target_link_libraries("${PROJECT_NAME}" PRIVATE "${PROJECT_NAME}_static")
